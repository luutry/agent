## 四、步骤 2：向量化脚本

创建文件 `D:\train-intent\01_embed_data.py`：

```python
"""
步骤1：将训练数据中的查询文本转为 Embedding 向量。
输入：data/train.jsonl（纯文本 + 标签）
输出：data/train_embedded.json（文本 + 标签 + 1024维向量）

运行方式：python 01_embed_data.py
"""

import json       # Python 内置 JSON 处理（类似 Java 的 Jackson）
import time       # 计时用（类似 Java 的 System.currentTimeMillis()）
import requests   # HTTP 客户端（类似 Java 的 OkHttp）

# ==================== 配置区（按需修改）====================

# DashScope API 配置（和项目中 application.yml 一致）
API_KEY = "sk-ws-H.PXLEHII.IrR8.MEUCIQDACVS-2Md9R80_Y6-c3Is8tWXOqtkFwapBA6VDmNIucQIgTvc0egzj7mhaf-uYj0V2mLaOioKt1Cdqn5y3XiBckfk"
BASE_URL = "https://dashscope.aliyuncs.com/compatible-mode/v1/embeddings"
MODEL = "text-embedding-v3"

# 文件路径
INPUT_FILE = "data/train.jsonl"
OUTPUT_FILE = "data/train_embedded.json"

# ==================== 核心函数 ====================

def call_embedding_api(text: str) -> list[float]:
    """
    调用 DashScope Embedding API，将文本转为 1024 维向量。
    类似 Java 中的：embeddingService.embed(text)
    
    参数：text - 要向量化的文本
    返回：float 列表，长度 1024
    """
    headers = {
        "Authorization": f"Bearer {API_KEY}",
        "Content-Type": "application/json"
    }
    payload = {
        "model": MODEL,
        "input": text
    }

    # 发送 HTTP POST 请求（类似 Java 的 httpClient.post(...)）
    response = requests.post(BASE_URL, headers=headers, json=payload, timeout=30)

    # 检查是否成功（类似 Java 的 if (response.getStatusCode() != 200)）
    if response.status_code != 200:
        raise Exception(f"API 调用失败: {response.status_code} - {response.text}")

    # 解析 JSON 响应（类似 Java 的 objectMapper.readTree(...)）
    result = response.json()
    embedding = result["data"][0]["embedding"]

    return embedding


def main():
    print(f"读取训练数据: {INPUT_FILE}")

    # 读取 JSONL 文件（每行一个 JSON 对象）
    samples = []
    with open(INPUT_FILE, "r", encoding="utf-8") as f:
        for line in f:
            line = line.strip()
            if line:  # 跳过空行
                samples.append(json.loads(line))  # json.loads 类似 Jackson 的 readValue

    print(f"共 {len(samples)} 条训练数据")

    # 逐条调用 Embedding API
    start_time = time.time()
    for i, sample in enumerate(samples):
        query = sample["query"]
        print(f"[{i+1}/{len(samples)}] 向量化: {query}")

        try:
            # 调用 API 获取向量
            vec = call_embedding_api(query)
            # 将向量存入样本
            sample["embedding"] = vec
            print(f"  → 成功，维度: {len(vec)}")
        except Exception as e:
            print(f"  → 失败: {e}")
            sample["embedding"] = None

        # API 限流保护：每次调用间隔 0.2 秒
        time.sleep(0.2)

    elapsed = time.time() - start_time
    print(f"\n向量化完成，耗时: {elapsed:.1f}秒")

    # 统计成功/失败数
    success = sum(1 for s in samples if s["embedding"] is not None)
    print(f"成功: {success}, 失败: {len(samples) - success}")

    # 保存结果（类似 Java 的 objectMapper.writerWithDefaultPrettyPrinter().writeValue(...)）
    with open(OUTPUT_FILE, "w", encoding="utf-8") as f:
        json.dump(samples, f, ensure_ascii=False, indent=2)

    print(f"结果已保存: {OUTPUT_FILE}")


# Python 的入口点（类似 Java 的 public static void main(String[] args)）
if __name__ == "__main__":
    main()
```

## 五、步骤 3：训练脚本

创建文件 `D:\train-intent\02_train.py`：

```python
"""
步骤2：用 Softmax 回归训练意图分类器。
输入：data/train_embedded.json（带向量的训练数据）
输出：控制台打印评估报告 + 内存中生成 W 矩阵和 b 向量

运行方式：python 02_train.py
"""

import json
import numpy as np
from sklearn.linear_model import LogisticRegression
from sklearn.model_selection import cross_val_score
from sklearn.metrics import classification_report, confusion_matrix

# ==================== 配置区 ====================

INPUT_FILE = "data/train_embedded.json"

# 类别顺序（必须与项目中 QueryIntent 枚举的索引顺序一致）
# 项目代码中 CLASS_INDEX = [TOOL, RAG, MIXED, CHITCHAT]
LABELS = ["TOOL", "RAG", "MIXED", "CHITCHAT"]

# ==================== 数据加载 ====================

def load_data(filepath: str):
    """
    加载带向量的训练数据。
    返回 X（矩阵，每行是一个 1024 维向量）和 y（数组，每个值是类别索引 0~3）。
    """
    with open(filepath, "r", encoding="utf-8") as f:
        samples = json.load(f)

    X = []  # 特征矩阵
    y = []  # 标签数组

    for sample in samples:
        # 跳过向量化失败的样本
        if sample["embedding"] is None:
            continue

        X.append(sample["embedding"])
        # 将文本标签转为数字索引：TOOL→0, RAG→1, MIXED→2, CHITCHAT→3
        label_index = LABELS.index(sample["label"])
        y.append(label_index)

    # 转为 numpy 数组（类似 Java 的 List<double[]> → double[][]）
    X = np.array(X)  # shape: [N, 1024]
    y = np.array(y)  # shape: [N]

    return X, y


# ==================== 训练 ====================

def train_model(X, y):
    """
    训练 Softmax 回归模型。
    
    参数说明：
    - C=10.0：正则化强度的倒数。C 越大越不过拟合，越小越容易过拟合。
              类似 Java 中没有直接等价物，可以理解为"模型复杂度控制"。
    - max_iter=1000：最大迭代次数，类似训练轮数。
    - multi_class="multinomial"：多分类模式（我们有 4 个类别）。
    - solver="lbfgs"：优化算法，适合中小数据集。
    """
    model = LogisticRegression(
        C=10.0,
        max_iter=1000,
        multi_class="multinomial",
        solver="lbfgs"
    )

    print("开始训练...")
    model.fit(X, y)
    print("训练完成！\n")

    return model


# ==================== 评估 ====================

def evaluate_model(model, X, y):
    """评估模型准确率并打印分类报告。"""

    # 5 折交叉验证（把数据分 5 份，轮流用 4 份训练、1 份测试，取平均准确率）
    # 类似 Java 中没有直接等价物，是 ML 领域的标准评估方法
    cv_scores = cross_val_score(model, X, y, cv=5, scoring="accuracy")
    print(f"5 折交叉验证准确率: {cv_scores.mean():.4f} ± {cv_scores.std():.4f}\n")

    # 在全部训练数据上预测
    y_pred = model.predict(X)

    # 分类报告（精确率、召回率、F1 值）
    print("分类报告:")
    print(classification_report(y, y_pred, target_names=LABELS))

    # 混淆矩阵（看哪些样本被分错了）
    print("混淆矩阵:")
    cm = confusion_matrix(y, y_pred)
    print(cm)
    print(f"\n矩阵含义（行=真实类别，列=预测类别）:")
    print(f"       预测TOOL  预测RAG  预测MIXED  预测CHITCHAT")
    for i, label in enumerate(LABELS):
        row = "  ".join(f"{cm[i][j]:>6}" for j in range(len(LABELS)))
        print(f"实际{label:>10}: {row}")

    return model


# ==================== 主流程 ====================

def main():
    # 加载数据
    print(f"加载数据: {INPUT_FILE}")
    X, y = load_data(INPUT_FILE)
    print(f"数据量: {X.shape[0]} 条, 向量维度: {X.shape[1]}")
    print(f"各类别分布: {dict(zip(LABELS, np.bincount(y, minlength=4)))}\n")

    # 训练
    model = train_model(X, y)

    # 评估
    model = evaluate_model(model, X, y)

    # 提取权重（下一步导出用）
    W = model.coef_       # shape: [4, 1024]，类似 Java 的 double[][]
    b = model.intercept_  # shape: [4]，类似 Java 的 double[]

    print(f"\n权重矩阵 W 维度: {W.shape}")
    print(f"偏置向量 b 维度: {b.shape}")
    print(f"\n提示：运行 03_export.py 导出为 YAML 配置文件")


if __name__ == "__main__":
    main()
```

## 六、步骤 4：导出脚本

创建文件 `D:\train-intent\03_export.py`：

```python
"""
步骤3：将训练好的权重矩阵导出为 YAML 格式，供 application.yml 直接使用。
输入：data/train_embedded.json（重新训练并导出，一步完成）
输出：output/weights.yml

运行方式：python 03_export.py
"""

import json
import yaml
import numpy as np
from sklearn.linear_model import LogisticRegression

# ==================== 配置区 ====================

INPUT_FILE = "data/train_embedded.json"
OUTPUT_FILE = "output/weights.yml"
LABELS = ["TOOL", "RAG", "MIXED", "CHITCHAT"]

# ==================== 主流程 ====================

def main():
    # 加载数据
    with open(INPUT_FILE, "r", encoding="utf-8") as f:
        samples = json.load(f)

    X, y = [], []
    for sample in samples:
        if sample["embedding"] is None:
            continue
        X.append(sample["embedding"])
        y.append(LABELS.index(sample["label"]))

    X = np.array(X)
    y = np.array(y)

    # 训练模型
    model = LogisticRegression(C=10.0, max_iter=1000, multi_class="multinomial", solver="lbfgs")
    model.fit(X, y)

    # 提取权重
    W = model.coef_       # [4, 1024]
    b = model.intercept_  # [4]

    # 转为 Python 原生 list（numpy 类型不能直接序列化为 YAML）
    weights_list = W.tolist()  # [[float, float, ...], ...] 4 行，每行 1024 个
    bias_list = b.tolist()     # [float, float, float, float]

    # 构建 YAML 配置
    config = {
        "weights": weights_list,
        "bias": [round(v, 6) for v in bias_list],  # 保留 6 位小数
        "metadata": {
            "trained_samples": len(X),
            "embedding_dim": X.shape[1],
            "labels": LABELS,
            "model": "LogisticRegression(multinomial)",
        }
    }

    # 写入文件
    with open(OUTPUT_FILE, "w", encoding="utf-8") as f:
        yaml.dump(config, f, default_flow_style=False, allow_unicode=True, sort_keys=False)

    print(f"导出完成: {OUTPUT_FILE}")
    print(f"训练样本数: {len(X)}")
    print(f"权重矩阵维度: {W.shape}")
    print(f"偏置向量: {bias_list}")

    # 同时输出精简版（只含权重和偏置，可直接粘贴到 application.yml）
    print(f"\n{'='*60}")
    print("将以下内容粘贴到 application.yml 的 agent.intent-classification 段：")
    print(f"{'='*60}\n")

    # 生成 application.yml 格式
    yml_content = "    weights:\n"
    for i, label in enumerate(LABELS):
        row = [round(v, 6) for v in weights_list[i]]
        yml_content += f"      # {label} 行（1024 维）\n"
        yml_content += f"      - [{', '.join(str(v) for v in row)}]\n"
    yml_content += f"    bias: [{', '.join(str(round(v, 6)) for v in bias_list)}]\n"

    # 写入精简版文件
    with open("output/application_yml_snippet.txt", "w", encoding="utf-8") as f:
        f.write(yml_content)

    print(yml_content[:500] + "...\n")
    print(f"完整内容已保存: output/application_yml_snippet.txt")


if __name__ == "__main__":
    main()
```

### 7.2 代码无需任何修改

`EmbeddingIntentClassifier` 的路径选择逻辑：

```java
if (hasWeights()) {
    // ← 填入 weights 后自动走这条路
    return classifyByLinearHead(queryEmbedding, query);
```

`hasWeights()` 判断：`weights.size() == 4 && weights.get(0).size() > 0`

- `weights: [[...], [...], [...], [...]]` → 走线性头路径

## 十、常见问题

### Q：运行 python 提示"不是内部命令"？
安装 Python 时没有勾选 "Add to PATH"。解决方法：重新运行 Python 安装程序，勾选该选项。

### Q：pip install 很慢？
使用国内镜像。**注意：`pip` 命令必须在 PowerShell 或 CMD 中执行，不能在 Python 交互环境（`>>>`）里输入**，否则会报 `SyntaxError: invalid syntax`。
```powershell
# 正确：在 PowerShell / CMD 中运行
pip install numpy scikit-learn requests pyyaml -i https://pypi.tuna.tsinghua.edu.cn/simple

# 错误：在 Python >>> 提示符下运行（会报 SyntaxError）
# >>> pip install numpy  ← 这是错的！
```

### Q：训练数据需要多少条才够？
最低 120 条（每类 30 条），推荐 200~400 条。不需要几千条——因为 Embedding 模型已经做了语义理解的重活，线性头只需要学一条分界线。

### Q：准确率到多少算合格？
交叉验证准确率 > 85% 即可上线。如果 < 80%，检查：
1. 训练数据是否有标注错误
2. 各类别样本是否严重不平衡
3. 混淆矩阵中哪类最差，针对性补充该类样本

### Q：能不能在 IDEA 里直接运行 Python？
可以安装 "Python" 插件（Community 版免费），但没必要。训练脚本就 3 个文件，命令行运行更简单直接。
